// SPDX-License-Identifier: MIT
// Part of kwinish (Copyright (c) 2026 lhen).
// The bar's host actions on Hyprland 0.57, replacing the 0.56.2 legacy dispatchers (g_pKeybindManager->m_dispatchers).
// Each wrapper keeps what the dispatcher translator did around the Config::Actions call (window selection, argument
// parsing, guards) and inspects the result: a failure is logged through a per-site gate (actionGate.hpp), never more
// than one line per site every FAILURE_LOG_WINDOW. A failure never changes what the caller does next, as before.
// Map and differences: docs/port-0-57/P8-DISPATCHERS.md.
#pragma once

#include "actionGate.hpp"

#include <hyprland/src/Compositor.hpp>
#include <hyprland/src/config/shared/actions/ConfigActions.hpp>
#include <hyprland/src/config/supplementary/executor/Executor.hpp>
#include <hyprland/src/debug/log/Logger.hpp>
#include <hyprland/src/desktop/state/FocusState.hpp>
#include <hyprland/src/desktop/view/window/Window.hpp>

#include <array>
#include <chrono>
#include <format>
#include <string>

namespace KwAct {
    inline std::array<SFailureGate, SITE_COUNT> g_failureGates{};

    // true if the action succeeded. A failure is logged at the level the host gave it (SILENT: not logged).
    inline bool check(eSite site, const Config::Actions::ActionResult& result) {
        if (result)
            return true;
        const auto& E = result.error();
        if (E.level == Config::eConfigErrorLevel::SILENT || site >= SITE_COUNT)
            return false;
        const auto SKIPPED = g_failureGates[site].admit(std::chrono::steady_clock::now());
        if (!SKIPPED)
            return false;
        const auto LEVEL = E.level == Config::eConfigErrorLevel::ERROR ? Log::ERR : E.level == Config::eConfigErrorLevel::WARNING ? Log::WARN : Log::DEBUG;
        if (*SKIPPED == 0)
            Log::logger->log(LEVEL, "kwinish-bars", "{} failed: {} ({})", siteName(site), E.message, Config::toString(E.code));
        else
            Log::logger->log(LEVEL, "kwinish-bars", "{} failed: {} ({}); {} earlier failure(s) not logged", siteName(site), E.message, Config::toString(E.code), *SKIPPED);
        return false;
    }

    // 0.56.2 resizewindowpixel "<cmd>,activewindow": the focused window (not necessarily the bar's), the size parsed
    // relative to its goal size, refused below 1x1.
    inline bool resizeFocused(eSite site, const std::string& cmd) {
        const auto W = Desktop::focusState()->window();
        if (!W)
            return check(site, Config::Actions::actionError("resizeWindow: no window", Config::eConfigErrorLevel::WARNING, Config::eConfigErrorCode::NO_TARGET));
        const auto SIZ = g_pCompositor->parseWindowVectorArgsRelative(cmd, W->size(Desktop::View::IGeometric::GEOMETRIC_GOAL));
        if (!usableSize(SIZ.x, SIZ.y))
            return check(site, Config::Actions::actionError("Invalid size", Config::eConfigErrorLevel::ERROR, Config::eConfigErrorCode::INVALID_ARGUMENT));
        return check(site, Config::Actions::resize(SIZ, false, W));
    }

    // 0.56.2 movewindowpixel "<cmd>,activewindow": the focused window, the position parsed relative to its goal position.
    inline bool moveFocused(eSite site, const std::string& cmd) {
        const auto W = Desktop::focusState()->window();
        if (!W)
            return check(site, Config::Actions::actionError("moveWindow: no window", Config::eConfigErrorLevel::WARNING, Config::eConfigErrorCode::NO_TARGET));
        const auto POS = g_pCompositor->parseWindowVectorArgsRelative(cmd, W->position(Desktop::View::IGeometric::GEOMETRIC_GOAL));
        return check(site, Config::Actions::move(POS, false, W));
    }

    // 0.56.2 exec: the executor's spawn, a failure being no pid or pid 0 (same messages as the old dispatcher).
    inline bool spawn(eSite site, const std::string& cmd) {
        const auto PROC = Config::Supplementary::executor()->spawn(cmd);
        if (spawnStarted(PROC))
            return true;
        return check(site,
                     Config::Actions::actionError(PROC ? std::format("Failed to start process {}", cmd) : std::format("Failed to start process. No closing bracket in exec rule. {}", cmd),
                                                  Config::eConfigErrorLevel::ERROR, Config::eConfigErrorCode::EXECUTION_FAILED));
    }

// SPDX-License-Identifier: MIT
// Part of kwinish (Copyright (c) 2026 lhen).
// The part of the bar's host actions that needs no compositor: the arguments and guards the 0.56.2 dispatcher
// translator applied, and the gate that keeps action-failure diagnostics from flooding the log. The Hyprland side
// is hostActions.hpp. Tested by tools/test-plugin-actions.sh.
#pragma once

#include <array>
#include <chrono>
#include <cstddef>
#include <cstdint>
#include <format>
#include <optional>
#include <string>
#include <string_view>

namespace KwAct {
    // Where a host action is called from; each site has its own failure-log gate.
    enum eSite : uint8_t {
        SITE_TOUCH_FLOAT = 0, // K4  touch drag start: float the focused window
        SITE_TOUCH_RESIZE,    // K5  touch drag start: exact 50% 50%
        SITE_TOUCH_PIN,       // K6  touch drag start: pin toggle
        SITE_TOUCH_MOVE,      // K7  touch drag: move to the finger
        SITE_TOUCH_TILE,      // K8  touch drag left the bar: tile again
        SITE_MAXIMIZE,        // K10 maximize button
        SITE_FULLSCREEN,      // K10 fullscreen button
        SITE_MINIMIZE,        // K11 minimize button
        SITE_EXEC_BUTTON,     // K12 exec:<cmd> button
        SITE_MENU,            // K12 window menu (kwinish-ui)
        SITE_COUNT,
    };

    inline std::string_view siteName(eSite s) {
        static constexpr std::array<std::string_view, SITE_COUNT> NAMES = {
            "touch float", "touch resize", "touch pin", "touch move", "touch tile", "maximize", "fullscreen", "minimize", "exec button", "window menu",
        };
        return s < SITE_COUNT ? NAMES[s] : "?";
    }

    // "exact X Y": the movewindowpixel argument the 0.56.2 call built, without the ",activewindow" selector.
    // The caller keeps its (int) casts so fractional coordinates are truncated exactly as before.
    inline std::string exactPixelArg(int x, int y) {
        return std::format("exact {} {}", x, y);
    }

    // The resizewindowpixel guard of the 0.56.2 translator, verbatim: refuse if SIZ.x < 1 || SIZ.y < 1 (NaN passes).
    inline bool usableSize(double x, double y) {
        return !(x < 1 || y < 1);
    }

    // The 0.56.2 exec dispatcher reported success as PROC.has_value() && PROC.value() > 0.
    inline bool spawnStarted(const std::optional<uint64_t>& pid) {
        return pid.has_value() && *pid > 0;
    }

    inline constexpr std::chrono::steady_clock::duration FAILURE_LOG_WINDOW = std::chrono::seconds(10);

    // One per call site. The first failure is logged; later ones inside FAILURE_LOG_WINDOW of the last logged line are
    // only counted, and the next logged line reports how many were skipped. Trivially destructible and holds no host
    // objects; whether the module itself can be unloaded safely is the separate unload work (Q4), not shown here.
    struct SFailureGate {
        std::optional<std::chrono::steady_clock::time_point> last;
        uint64_t                                             suppressed = 0;

        // nullopt: do not log this one. Otherwise log it, together with the returned number of skipped failures.
        std::optional<uint64_t> admit(std::chrono::steady_clock::time_point now) {
            if (last && now < *last + FAILURE_LOG_WINDOW) {
                ++suppressed;
                return std::nullopt;
            }
            const uint64_t SKIPPED = suppressed;
            last                   = now;
            suppressed             = 0;
            return SKIPPED;
        }
    };
}

// SPDX-License-Identifier: MIT
// Part of kwinish (Copyright (c) 2026 lhen).
// Tests for actionGate.hpp (no Hyprland): bash tools/test-plugin-actions.sh
#include "actionGate.hpp"

#include <chrono>
#include <cmath>
#include <cstdio>
#include <limits>
#include <optional>
#include <string>

static int fails = 0;

static void check(bool okay, const std::string& what) {
    std::printf("%s - %s\n", okay ? "ok" : "FALHOU", what.c_str());
    fails += okay ? 0 : 1;
}

int main() {
    using namespace std::chrono_literals;
    using clk = std::chrono::steady_clock;

    // K7: the pixel argument the 0.56.2 call built, minus ",activewindow"; the (int) casts stay at the call site
    check(KwAct::exactPixelArg(10, 20) == "exact 10 20", "exact pixel argument");
    check(KwAct::exactPixelArg(-3, 0) == "exact -3 0", "negative coordinate kept");
    check(KwAct::exactPixelArg((int)-3.7, (int)7.9) == "exact -3 7", "casts truncate toward zero, as before");

    // K5: the translator's guard (resizewindowpixel): SIZ.x < 1 || SIZ.y < 1 refuses
    check(KwAct::usableSize(1, 1), "1x1 is usable");
    check(KwAct::usableSize(800, 600), "800x600 is usable");
    check(!KwAct::usableSize(0.999, 600), "width below 1 refused");
    check(!KwAct::usableSize(800, 0), "height 0 refused");
    check(!KwAct::usableSize(-5, -5), "negative refused");
    const double NAN_ = std::numeric_limits<double>::quiet_NaN();
    check(KwAct::usableSize(NAN_, 600), "NaN passes, exactly like the translator's comparison");

    // K12: the 0.56.2 exec dispatcher reported success = PROC.has_value() && PROC.value() > 0
    check(!KwAct::spawnStarted(std::nullopt), "spawn nullopt is a failure");
    check(!KwAct::spawnStarted(std::optional<uint64_t>{0}), "spawn pid 0 is a failure");
    check(KwAct::spawnStarted(std::optional<uint64_t>{4242}), "spawn pid > 0 is a success");

    // failure diagnostics: first failure logged, then at most one line per window per site, with the count skipped
    {
        KwAct::SFailureGate g;
        const auto          T0 = clk::time_point{} + 1000s;
        auto                a  = g.admit(T0);
        check(a.has_value() && *a == 0, "first failure is logged, nothing suppressed");
        check(!g.admit(T0 + 1s).has_value(), "second failure inside the window is suppressed");
        check(!g.admit(T0 + 9s).has_value(), "third failure inside the window is suppressed");
        a = g.admit(T0 + KwAct::FAILURE_LOG_WINDOW);
        check(a.has_value() && *a == 2, "after the window: logged, reports 2 suppressed");
        check(!g.admit(T0 + KwAct::FAILURE_LOG_WINDOW + 1s).has_value(), "window restarts at the logged line");
        a = g.admit(T0 + 3 * KwAct::FAILURE_LOG_WINDOW);
        check(a.has_value() && *a == 1, "suppressed count resets after each logged line");
    }
    {
        // a clock that goes backwards (it cannot for steady_clock, but the gate must not lock up forever)
        KwAct::SFailureGate g;
        const auto          T0 = clk::time_point{} + 1000s;
        (void)g.admit(T0);
        check(!g.admit(T0 - 5s).has_value(), "earlier timestamp inside the window is suppressed");
        check(g.admit(T0 + KwAct::FAILURE_LOG_WINDOW).has_value(), "and the gate still opens after the window");
    }
    {
        // gates are independent per site
        KwAct::SFailureGate a, b;
        const auto          T0 = clk::time_point{} + 1000s;
        (void)a.admit(T0);
        check(b.admit(T0 + 1s).has_value(), "a second site is not suppressed by the first");
    }
    check(KwAct::FAILURE_LOG_WINDOW == 10s, "window is 10 s");

    // site names are fixed, distinct and non-empty
    bool namesOk = true;
    for (size_t i = 0; i < KwAct::SITE_COUNT; ++i) {
        const std::string_view n = KwAct::siteName((KwAct::eSite)i);
        namesOk                  = namesOk && !n.empty() && n != "?";
        for (size_t j = 0; j < i; ++j)
            namesOk = namesOk && n != KwAct::siteName((KwAct::eSite)j);
    }
    check(namesOk, "every site has a distinct name");
    check(KwAct::siteName(KwAct::SITE_COUNT) == "?", "out-of-range site is '?'");

    return fails > 0 ? 1 : 0;
}

#!/usr/bin/env python3
"""P8 extraction.

extract.py calls SRC OUT MODE   MODE=old: plugin lines using m_dispatchers[...] / dispatchIfKnown(...) (K4-K8, K10-K12;
                                the K9 line `["mouse"]("0movewindow")` is out of scope and skipped);
                                MODE=new: plugin lines containing `KwAct::` (statements, not includes/comments).
extract.py old-fn SRC OUT       the plugin's own dispatchIfKnown definition, verbatim.
extract.py translator OSRC OUT  verbatim 0.56.2 DispatcherTranslator helpers + the 8 dispatchers the plugin used.
Each extracted plugin statement becomes `case K:` with a #line back to the real file.
"""
import re, sys

EXPECT = 10
ORDER = ["K4", "K5", "K6", "K7", "K8", "K10a", "K10b", "K11", "K12a", "K12b"]


def stmts(lines, pred):
    out, i, in_def = [], 0, False
    while i < len(lines):
        l = lines[i]
        if l.startswith("static void dispatchIfKnown("):
            in_def = True
        if in_def:
            if l.startswith("}"):
                in_def = False
            i += 1
            continue
        s = l.strip()
        if pred(l) and not s.startswith("//") and not s.startswith("#"):
            j = i
            while not lines[j].split("//")[0].rstrip().endswith(";"):
                j += 1
                assert j - i < 6, f"unterminated at {i+1}"
            out.append((i + 1, lines[i : j + 1]))
            i = j + 1
        else:
            i += 1
    return out


def main():
    mode = sys.argv[1]
    if mode == "calls":
        src, out, kind = sys.argv[2:5]
        lines = open(src, encoding="utf-8").read().split("\n")
        if kind == "old":
            pred = lambda l: ("m_dispatchers[" in l and '"mouse"' not in l) or "dispatchIfKnown(" in l
        else:
            pred = lambda l: "KwAct::" in l
        st = stmts(lines, pred)
        assert len(st) == EXPECT, f"{kind}: expected {EXPECT} statements, found {len(st)} at {[s[0] for s in st]}"
        def enclosing(ln):  # every enclosing control line, innermost first, up to the function header
            chain, ind = [], len(lines[ln - 1]) - len(lines[ln - 1].lstrip())
            for n in range(ln - 2, -1, -1):
                s = lines[n].strip()
                if s and not s.startswith("//") and len(lines[n]) - len(lines[n].lstrip()) < ind:
                    chain.append(s)
                    ind = len(lines[n]) - len(lines[n].lstrip())
                    if ind == 0:
                        break
            return " <- ".join(chain) or "?"
        with open(out + ".guards", "w") as g:
            for k, (ln, _) in enumerate(st):
                g.write(f"{ORDER[k]}: {enclosing(ln)}\n")
        with open(out, "w") as f:
            for k, (ln, body) in enumerate(st):
                f.write(f"    case {k}: {{ // {ORDER[k]}\n#line {ln} \"{src}\"\n" + "\n".join(body) + "\n    } break;\n")
        print(" ".join(f"{ORDER[k]}:{ln}" for k, (ln, _) in enumerate(st)))
    elif mode == "old-fn":
        src, out = sys.argv[2:4]
        lines = open(src, encoding="utf-8").read().split("\n")
        i = next(n for n, l in enumerate(lines) if l.startswith("static void dispatchIfKnown("))
        j = next(n for n in range(i, len(lines)) if lines[n].startswith("}"))
        open(out, "w").write(f"#line {i+1} \"{src}\"\n" + "\n".join(lines[i : j + 1]) + "\n")
        print(f"dispatchIfKnown {i+1}-{j+1}")
    elif mode == "translator":
        src, out = sys.argv[2:4]
        t = open(src, encoding="utf-8").read().split("\n")
        want = ["wrap", "windowFromArg", "resolveWorkspace", "exec", "setfloating", "settiled", "fullscreen",
                "movetoworkspacesilent", "movewindowpixel", "resizewindowpixel", "pin"]
        parts = []
        for name in want:
            rx = re.compile(r"^static [\w:<>]+ " + name + r"\(")
            i = next(n for n, l in enumerate(t) if rx.match(l))
            j = next(n for n in range(i, len(t)) if t[n].startswith("}"))
            parts.append(f"#line {i+1} \"{src}\"\n" + "\n".join(t[i : j + 1]))
        # CDispatcherTranslator::run, verbatim body (as a free function over a local map)
        i = next(n for n, l in enumerate(t) if l.startswith("SDispatchResult CDispatcherTranslator::run("))
        j = next(n for n in range(i, len(t)) if t[n].startswith("}"))
        run = "\n".join(t[i : j + 1]).replace("SDispatchResult CDispatcherTranslator::run(", "SDispatchResult translatorRun(")
        reg = []
        for n, l in enumerate(t):
            m = re.match(r'^\s*m_dispMap\["(\w+)"\]\s*=\s*::(\w+);', l)
            if m and m.group(1) in want:
                reg.append(f'#line {n+1} "{src}"\n' + l)
        assert len(reg) == 8, reg
        open(out, "w").write("\n\n".join(parts) + "\n\n"
                             "inline std::unordered_map<std::string, std::function<SDispatchResult(const std::string&)>> m_dispMap;\n"
                             + f"#line {i+1} \"{src}\"\n" + run + "\n"
                             "static void registerDispatchers() {\n" + "\n".join(reg) + "\n}\n")
        print("translator: " + " ".join(want) + f"; run {i+1}; {len(reg)} registrations")


main()

#!/usr/bin/env python3
"""P8 log-policy oracle for the NEW side ('logs' mode: every scenario runs the statement 3 times in a row).
Expected, computed from the scenario alone (not from the implementation):
  K5/K7 with no focused window      -> 1 WARN  '<site> failed: <...>: no window (no_target)'
  K5 with a parse below 1x1          -> 1 ERR   'Invalid size (invalid_argument)'
  K12 with spawn pid 0 / no pid      -> 1 ERR   'Failed to start process ...' (execution_failed)
  otherwise a failed action result   -> info: 1 DEBUG, warn: 1 WARN, err: 1 ERR; silent/ok: none
Ordinary action failures must carry the host message ('fake <action> <info|warning|error>').
Three immediate repeats give exactly one line (the gate); after the window is rewound a 4th call gives a second line
ending '; 2 earlier failure(s) not logged'. No line when nothing failed."""
import re, sys

SITES = ["touch float", "touch resize", "touch pin", "touch move", "touch tile", "maximize", "fullscreen", "minimize", "exec button", "window menu"]
LEVEL = {"info": "DEBUG", "warn": "WARN", "err": "ERR"}
CODE = {"info": "invalid_state", "warn": "invalid_state", "err": "invalid_argument"}

blocks, cur = [], None
for line in open(sys.argv[1], encoding="utf-8"):
    line = line.rstrip("\n")
    m = re.match(r"^k=(\d+) focus=(\w+) res=(\w+) parse=(\w+) spawn=(\w+) \| (.*)$", line)
    if m:
        cur = {"k": int(m[1]), "focus": m[2], "res": m[3], "parse": m[4], "spawn": m[5], "calls": m[6], "logs": []}
        blocks.append(cur)
    elif line.startswith("  LOG"):
        cur["logs"].append(line.strip())
    else:
        print("unexpected line:", line); sys.exit(1)

bad = 0
for b in blocks:
    k, site = b["k"], SITES[b["k"]]
    exp = None  # (level, substring)
    if k in (8, 9):
        cmd = "kitty --title 'a b'" if k == 8 else "kwinish-ui menu <bar-window> --at 120,80"
        if b["spawn"] == "zero":
            exp = ("ERR", f"failed: Failed to start process {cmd} (execution_failed)")
        elif b["spawn"] == "none":
            exp = ("ERR", f"failed: Failed to start process. No closing bracket in exec rule. {cmd} (execution_failed)")
    elif k in (1, 3) and b["focus"] == "none":
        exp = ("WARN", ("resizeWindow" if k == 1 else "moveWindow") + ": no window (no_target)")
    elif k == 1 and b["parse"] == "tiny":
        exp = ("ERR", "Invalid size (invalid_argument)")
    elif b["res"] in LEVEL:
        fn = b["calls"].split(" ; ")[-1].split("(")[0]
        word = {"info": "info", "warn": "warning", "err": "error"}[b["res"]]
        exp = (LEVEL[b["res"]], f"fake {fn} {word} ({CODE[b['res']]})")
    logs = b["logs"]
    if exp is None:
        if logs:
            bad += 1; print("unexpected log:", b)
        continue
    if len(logs) != 2:
        bad += 1; print(f"expected exactly 2 log lines (first + summary), got {len(logs)}:", b); continue
    want = f"LOG[{exp[0]}] [kwinish-bars] {site} failed: "
    if not logs[0].startswith(want) or exp[1] not in logs[0] or "earlier failure" in logs[0]:
        bad += 1; print("wrong first log line:", logs[0], "| want prefix", want, "and", exp[1])
    if not logs[1].startswith(want) or exp[1] not in logs[1] or not logs[1].endswith("; 2 earlier failure(s) not logged"):
        bad += 1; print("wrong summary line:", logs[1])
    if k in (8, 9) and b["spawn"] in ("zero", "none") and "(execution_failed)" not in logs[0]:
        bad += 1; print("spawn failure without execution_failed code:", logs[0])
n_logged = sum(1 for b in blocks if b["logs"])
print(f"log oracle: {len(blocks)} scenarios, {n_logged} with first+summary lines, {bad} mismatches")
sys.exit(1 if bad or len(blocks) != 1350 else 0)

#!/usr/bin/env python3
"""P8 diff audit: every line changed in barDeco.cpp (BASE -> CAND) must be one of
  removed: an old extracted dispatcher statement, a line of the dispatchIfKnown definition (by position, + the blank
           line after it), a comment;
  added:   a new extracted KwAct statement, the hostActions.hpp include, a comment.
Within one hunk, an identical line removed and re-added (diff alignment, e.g. an unchanged `}`) cancels out.
Anything else (a new `return;`, a changed condition, a moved statement) fails, so the extracted statements stay
reachable exactly as before.  usage: diff_audit.py BASE CAND OLD_CALLS NEW_CALLS"""
import collections, difflib, sys

base, cand, oldc, newc = sys.argv[1:5]
b = open(base, encoding="utf-8").read().split("\n")
c = open(cand, encoding="utf-8").read().split("\n")


def stmt_lines(inc):
    return {l.rstrip("\n").strip() for l in open(inc, encoding="utf-8") if not l.startswith(("    case ", "#line", "    } break;"))}


old_ok, new_ok = stmt_lines(oldc), stmt_lines(newc)
i0 = next(n for n, l in enumerate(b) if l.startswith("static void dispatchIfKnown("))
j0 = next(n for n in range(i0, len(b)) if b[n].startswith("}"))
def_range = set(range(i0, j0 + 2 if b[j0 + 1].strip() == "" else j0 + 1))
bad, counts = 0, collections.Counter()
for tag, i1, i2, j1, j2 in difflib.SequenceMatcher(None, b, c, autojunk=False).get_opcodes():
    if tag == "equal":
        continue
    rem = [(n, b[n].strip()) for n in range(i1, i2)]
    add = [c[n].strip() for n in range(j1, j2)]
    generic_rem, generic_add = collections.Counter(), collections.Counter()
    for n, s in rem:
        if s in old_ok: counts["-stmt"] += 1
        elif n in def_range: counts["-def"] += 1
        elif s.startswith("//"): counts["-comment"] += 1
        else: generic_rem[s] += 1
    for s in add:
        if s in new_ok: counts["+stmt"] += 1
        elif s == '#include "hostActions.hpp"': counts["+include"] += 1
        elif s.startswith("//"): counts["+comment"] += 1
        else: generic_add[s] += 1
    moved = generic_rem & generic_add
    counts["realigned"] += sum(moved.values())
    for s, k in list((generic_rem - moved).items()) + list((generic_add - moved).items()):
        bad += k
        print(f"UNEXPLAINED change in hunk base {i1+1}-{i2} / cand {j1+1}-{j2}: '{s[:120]}'")
print("diff audit:", dict(counts), "unexplained:", bad)
sys.exit(1 if bad or counts["-stmt"] != 10 or counts["+stmt"] != 10 else 0)

    if (!ws)
        return actionError("No workspace to move to", eActionErrorLevel::WARNING, eActionErrorCode::INVALID_ARGUMENT);

    if (ws == window->m_workspace)
        return {};

    auto       constraintGuard = g_pInputManager->pointerConstraints().suspend();

    const auto POLDWS = window->m_workspace;

    updateRelativeCursorCoords();
    g_pHyprRenderer->damageWindow(window);

    if (silent) {
        const auto OLDMIDDLE = window->middle();
        Desktop::globalWindowController()->moveWindowToWorkspace(window, ws);

        if (window->m_workspace != POLDWS && window == Desktop::focusState()->window()) {
            if (const auto PATCOORDS =
                    Desktop::viewState()->hitTest().windowAt(OLDMIDDLE, Desktop::View::RESERVED_EXTENTS | Desktop::View::INPUT_EXTENTS | Desktop::View::ALLOW_FLOATING, window);
                PATCOORDS)
                Desktop::focusState()->fullWindowFocus(PATCOORDS, Desktop::FOCUS_REASON_KEYBIND);
            else {
                // try to focus anything. If there isn't anything, this will unfocus (nullptr)
                const auto PATWS = POLDWS->getFirstWindow();
                Desktop::focusState()->fullWindowFocus(PATWS, Desktop::FOCUS_REASON_KEYBIND);
            }
        }
    } else {
        PHLMONITOR pMonitor = nullptr;

        const auto FULLSCREENMODE = Fullscreen::controller()->getFullscreenModes(window).internal;
        Desktop::globalWindowController()->moveWindowToWorkspace(window, ws);
        pMonitor = ws->m_monitor.lock();
        Desktop::focusState()->rawMonitorFocus(pMonitor);
        Fullscreen::controller()->setFullscreenMode(window, FULLSCREENMODE);

        POLDWS->rememberFocusedWindow(POLDWS->getFirstWindow());

        if (ws->type() == Workspace::eWorkspaceType::SPECIAL)
            pMonitor->setSpecialWorkspace(ws);
        else if (POLDWS->type() == Workspace::eWorkspaceType::SPECIAL)
            POLDWS->m_monitor.lock()->setSpecialWorkspace(nullptr);

        pMonitor->changeWorkspace(ws);

        Desktop::focusState()->fullWindowFocus(window, Desktop::FOCUS_REASON_KEYBIND);
        window->warpCursor();
    }

    return {};
}

    } else if (ARGS[1] == "unset") {

            return wrap(Actions::fullscreenWindow(MODE, true));
        return {};
    }

    return {};
}
