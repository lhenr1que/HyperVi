  570  void CHyprBar::stepUnsnap() {
  571      const auto W = m_pWindow.lock();
  572      if (!m_unsnapAnim || !W || !m_bDraggingThis) {
  573          m_unsnapAnim = false;
  574          return;
  575      }

  578      const Vector2D SZ  = m_unsnapFrom + (m_unsnapTo - m_unsnapFrom) * E;
  579      const auto     C   = g_pInputManager->getMouseCoordsInternal();

  582      (void)Config::Actions::move(POS, false, W);

  584          m_unsnapAnim = false;
  585          m_dragStart  = POS;
  586          g_pKeybindManager->changeMouseBindMode(MBIND_INVALID);
  587          g_pKeybindManager->changeMouseBindMode(MBIND_MOVE);
