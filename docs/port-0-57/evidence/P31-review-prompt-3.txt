// SPDX-License-Identifier: MIT
// Part of kwinish (Copyright (c) 2026 lhen).
#pragma once

#include <algorithm>
#include <cmath>
#include <optional>

namespace KwGeometry {
    struct LogicalPoint {
        double x = 0, y = 0;
    };
    struct LogicalBox {
        double x = 0, y = 0, w = 0, h = 0;
    };
    struct PhysicalBox {
        double x = 0, y = 0, w = 0, h = 0;
    };

    // Supported numeric domain (closed bounds). Every function returns
    // std::nullopt, never a wrong box, for NaN/inf, a bound violation or a
    // non-positive length/scale. Rationale, margins and measurements:
    // docs/port-0-57/GEOMETRY.md section 4.
    //  - kMaxLogical: |component| of every raw logical input (D, E, W, F, M,
    //    extBottom, aaMargin) and the limit of D.w/D.h: Wayland positions and
    //    sizes are int32, so 2^31 covers every value the compositor receives.
    //  - kMin/kMaxScale: monitor scale is a float; 1/16..16 is far wider than
    //    any real output and keeps pixel magnitudes <= 2^44.
    //  - kMaxPhysical: |component| of a preview box (kMaxLogical * kMaxScale).
    //  - kMaxBox: |component| of a derived logical box fed to unionDamage; it
    //    also bounds previewBBox's result (checked by static_assert in the test).
    // Coverage was measured to fail only from ~2^50 pixels, where one ulp of the
    // logical coordinate reaches a quarter pixel; the largest magnitude here is
    // 2^44, so no coverage depends on rounding slack of the representation.
    // Edges are the double values x+w / y+h, exactly what the host computes for
    // a CBox; an extent too small to move its own origin in double arithmetic
    // (x+w == x) is degenerate and rejected as well (see detail::resolved).
    // Every LogicalBox returned (bbox, union, preview) is itself valid input to
    // unionDamage (detail::boxOk); a result outside that box domain is nullopt.
    inline constexpr double kMaxLogical = 0x1p31, kMaxPhysical = 0x1p35, kMaxBox = 0x1p40;
    inline constexpr double kMinScale = 0x1p-4, kMaxScale = 0x1p4;

    namespace detail {
        // All predicates are false for NaN.
        inline bool within(double v, double limit) { return std::fabs(v) <= limit; }
        inline bool length(double v, double limit) { return v > 0 && v <= limit; }
        inline bool nonneg(double v, double limit) { return v >= 0 && v <= limit; }
        inline bool scaleOk(double s) { return s >= kMinScale && s <= kMaxScale; }
        // The extent survives addition to its origin; false would let the edge
        // collapse onto the origin (and, in physicalLocal, a residual of -1).
        inline bool resolved(double x, double w) { return x + w > x; }
        inline bool pointOk(LogicalPoint p) { return within(p.x, kMaxLogical) && within(p.y, kMaxLogical); }
        inline bool logicalOk(LogicalBox d, LogicalPoint e, LogicalPoint w, LogicalPoint f, LogicalPoint m) {
            return within(d.x, kMaxLogical) && within(d.y, kMaxLogical) && length(d.w, kMaxLogical) &&
                length(d.h, kMaxLogical) && pointOk(e) && pointOk(w) && pointOk(f) && pointOk(m);
        }
        inline bool boxOk(LogicalBox b) {
            return within(b.x, kMaxBox) && within(b.y, kMaxBox) && length(b.w, kMaxBox) && length(b.h, kMaxBox) &&
                resolved(b.x, b.w) && resolved(b.y, b.h);
        }
    }

    // D is relative logical geometry; E is the global logical anchor; W/F are
    // presentation offsets; M is the render monitor's global logical origin.
    // Domain: see the constants above. No host projection/rotation/renderModif
    // is applied. Match Hyprutils CBox::round: origin first, then residual
    // extent, including std::round's ties away from zero. Rounding w/h
    // independently is incorrect.
    inline std::optional<PhysicalBox> physicalLocal(LogicalBox d, LogicalPoint e, LogicalPoint w,
                                                    LogicalPoint f, LogicalPoint m, double scale) {
        if (!detail::logicalOk(d, e, w, f, m) || !detail::scaleOk(scale)) return std::nullopt;
        const double x = (d.x + e.x + w.x + f.x - m.x) * scale;
        const double y = (d.y + e.y + w.y + f.y - m.y) * scale;
        if (!detail::resolved(x, d.w * scale) || !detail::resolved(y, d.h * scale)) return std::nullopt;
        const double rx = std::round(x), ry = std::round(y);
        return PhysicalBox{rx, ry, std::round(x + d.w * scale - rx), std::round(y + d.h * scale - ry)};
    }

    // Logical monitor-local bounds. extBottom (e.g. ROUNDING*3) and aaMargin
    // are logical lengths in [0, kMaxLogical]; do not pass physical pixels.
    inline std::optional<LogicalBox> logicalBBox(LogicalBox d, LogicalPoint e, LogicalPoint w,
                                                 LogicalPoint f, LogicalPoint m, double extBottom, double aaMargin) {
        if (!detail::logicalOk(d, e, w, f, m) || !detail::nonneg(extBottom, kMaxLogical) ||
            !detail::nonneg(aaMargin, kMaxLogical))
            return std::nullopt;
        const LogicalBox b{d.x + e.x + w.x + f.x - m.x - aaMargin,
                           d.y + e.y + w.y + f.y - m.y - aaMargin,
                           d.w + 2 * aaMargin, d.h + extBottom + 2 * aaMargin};
        return detail::boxOk(b) ? std::optional<LogicalBox>(b) : std::nullopt;
    }

    // Both boxes must be in the domain (|x|,|y| <= kMaxBox, 0 < w,h <= kMaxBox,
    // x+w > x and y+h > y),
    // logical and in the SAME monitor-local coordinate system. This returns a
    // conservative enclosing rectangle, not a region. The caller owns
    // conversion if the monitor changes.
    inline std::optional<LogicalBox> unionDamage(LogicalBox prev, LogicalBox next) {
        if (!detail::boxOk(prev) || !detail::boxOk(next)) return std::nullopt;
        const double x = std::min(prev.x, next.x), y = std::min(prev.y, next.y);
        const double right = std::max(prev.x + prev.w, next.x + next.w);
        const double bottom = std::max(prev.y + prev.h, next.y + next.h);
        double w = right - x, h = bottom - y;
        // Subtraction followed by addition can lose a fractional boundary.
        if (x + w < right) w = std::nextafter(w, INFINITY);
        if (y + h < bottom) h = std::nextafter(h, INFINITY);
        // Never return a box this function would itself refuse as input.
        const LogicalBox result{x, y, w, h};
        return detail::boxOk(result) ? std::optional<LogicalBox>(result) : std::nullopt;
    }

    // Preview is already screen-space physical monitor-local geometry
    // (|x|,|y| <= kMaxPhysical, 0 < w,h <= kMaxPhysical, resolvable extents). Do not add W/F/M
    // again. Divide once, preserving fractions. If floating point cancellation
    // would lose coverage on floor/ceil after rescaling, widen only the
    // affected boundary by directed representable rounding. Use ordinary IEEE
    // arithmetic (not -ffast-math).
    inline std::optional<LogicalBox> previewBBox(PhysicalBox physical, double scale) {
        if (!detail::within(physical.x, kMaxPhysical) || !detail::within(physical.y, kMaxPhysical) ||
            !detail::length(physical.w, kMaxPhysical) || !detail::length(physical.h, kMaxPhysical) ||
            !detail::resolved(physical.x, physical.w) || !detail::resolved(physical.y, physical.h) ||
            !detail::scaleOk(scale))
            return std::nullopt;
        LogicalBox b{physical.x / scale, physical.y / scale, physical.w / scale, physical.h / scale};
        const auto coverAxis = [scale](double start, double size, double& origin, double& extent) {
            if (std::floor(origin * scale) > start)
                origin = std::nextafter(origin, -INFINITY);
            const double end = start + size;
            if (std::ceil((origin + extent) * scale) < end) {
                const double boundary = std::nextafter(end / scale, INFINITY);
                extent = std::nextafter(boundary - origin, INFINITY);
            }
        };
        coverAxis(physical.x, physical.w, b.x, b.w);
        coverAxis(physical.y, physical.h, b.y, b.h);
        // A result whose extent collapsed (division underflow, sub-ulp sliver) or
        // left the box domain is refused instead of returned.
        return detail::boxOk(b) ? std::optional<LogicalBox>(b) : std::nullopt;
    }
}

            check(pv && covers(outward(*pv, 1), phys(1, 0x1p-52)), "P5 preview of the smallest resolvable extent keeps coverage");
        }

        // Every field: every non-finite / overflow / just-outside value is rejected.
        for (int i = 0; i < kFields; ++i) {
            const double lim = i == 12 ? kMaxScale : kMaxLogical;
            for (double bad : {double(NAN), double(INFINITY), -double(INFINITY), DBL_MAX, -DBL_MAX, 0x1p53, -0x1p53,
                               std::nextafter(lim, INFINITY), -std::nextafter(lim, INFINITY)}) {
                if (i != 12 && !isLength(i) && std::abs(bad) <= lim) continue;
                Args a = base;
                field(a, i) = bad;
                check(!physicalOk(a), "P3 physicalLocal rejects non-finite/overflow/out-of-domain field");
                if (i != 12) check(!bboxOk(a), "P3 logicalBBox rejects non-finite/overflow/out-of-domain field");
            }
            // Lower edge of the positive quantities.
            if (isLength(i) || i == 12) {
                std::initializer_list<double> badLow = i == 12
                    ? std::initializer_list<double>{0, -1, -0., std::nextafter(kMinScale, 0), 0x1p-1074}
                    : std::initializer_list<double>{0, -0., -1, -0x1p-1074};
                for (double bad : badLow) {
                    Args a = base;
                    field(a, i) = bad;
                    check(!physicalOk(a), "P3 physicalLocal rejects zero/negative length or scale below minimum");
                    if (i != 12) check(!bboxOk(a), "P3 logicalBBox rejects zero/negative length");
                    // AA and extension must not make a nonpositive D length acceptable.
                    if (i != 12) check(!bboxOk(a, 5, 5), "P3 logicalBBox rejects zero/negative length even when AA/extension would widen it");
                }
            }
            // Exactly at the limit: accepted (the bound is closed).
            for (double good : {lim, i == 12 || isLength(i) ? kMinScale : -lim}) {
                Args a = base;
                field(a, i) = good;
                check(physicalOk(a), "P3 physicalLocal accepts the closed domain limit");
                if (i != 12) check(bboxOk(a), "P3 logicalBBox accepts the closed domain limit");
            }
        }
        // extBottom / aaMargin: finite, nonnegative, <= kMaxLogical.
        for (double bad : {double(NAN), double(INFINITY), DBL_MAX, -1., -0x1p-1074, std::nextafter(kMaxLogical, INFINITY)}) {
            check(!bboxOk(base, bad, 0), "P3 logicalBBox rejects bad extBottom");
            check(!bboxOk(base, 0, bad), "P3 logicalBBox rejects bad aaMargin");
        }
        check(bboxOk(base, kMaxLogical, kMaxLogical) && bboxOk(base, 0, 0), "P3 logicalBBox accepts extBottom/aaMargin at 0 and at the limit");
        check(!bboxOk(base, -0.0 - 1e-300, 0), "P3 logicalBBox rejects tiny negative extBottom");

        // Scale bounds for the other two scale-bearing functions.
        const PhysicalBox ph{10, 20, 30, 40};
        for (double bad : {double(NAN), double(INFINITY), DBL_MAX, 0., -1., 0x1p-1074, std::nextafter(kMinScale, 0), std::nextafter(kMaxScale, INFINITY)})
            check(!previewBBox(ph, bad).has_value(), "P5 previewBBox rejects bad scale");
        for (double good : {kMinScale, kMaxScale, 1.5})
            check(previewBBox(ph, good).has_value(), "P5 previewBBox accepts closed scale limits");
        // Preview physical box bounds.
        const double lim[]{kMaxPhysical, kMaxPhysical, kMaxPhysical, kMaxPhysical};
        for (int i = 0; i < 4; ++i)
            for (double bad : {double(NAN), double(INFINITY), -double(INFINITY), DBL_MAX, 0x1p53, std::nextafter(lim[i], INFINITY),
                               -std::nextafter(lim[i], INFINITY), (i >= 2 ? 0. : 1e300), (i >= 2 ? -1. : 1e300)}) {

                double v[]{10, 20, 30, 40};
                v[i] = bad;
                check(!previewBBox({v[0], v[1], v[2], v[3]}, 1.5).has_value(), "P5 previewBBox rejects bad physical component");
            }
        // unionDamage components: |x|,|y| <= kMaxBox, 0 < w,h <= kMaxBox.
        for (int i = 0; i < 4; ++i)
            for (double bad : {double(NAN), double(INFINITY), -double(INFINITY), DBL_MAX, 0x1p53, std::nextafter(kMaxBox, INFINITY),
                               -std::nextafter(kMaxBox, INFINITY), 0., -1.}) {

                double v[]{1, 2, 3, 4};
                v[i] = bad;
                const LogicalBox odd{v[0], v[1], v[2], v[3]}, ok{1, 2, 3, 4};
                check(!unionDamage(odd, ok).has_value() && !unionDamage(ok, odd).has_value(), "P4 unionDamage rejects bad component in either operand");

        check(!unionDamage({-kMaxBox, -kMaxBox, kMaxBox, kMaxBox}, {kMaxBox, kMaxBox, kMaxBox, kMaxBox}).has_value(), "P4 unionDamage refuses a union wider than the box domain");
        check(!unionDamage({-kMaxBox, -kMaxBox, kMaxBox / 2, kMaxBox / 2}, {-kMaxBox / 2, -kMaxBox / 2, std::nextafter(kMaxBox / 2, INFINITY), kMaxBox / 2}).has_value(), "P4 unionDamage refuses a union one ulp wider than kMaxBox");
        // previewBBox results must be valid boxes (reviewer: collapsed extents).
        check(!previewBBox({1.5, 0, 3 * 0x1p-54, 1}, 1.5).has_value(), "P5 previewBBox refuses a result whose extent collapsed by division");
        check(!previewBBox({-0x1p-1074, 0, 0x1p-1074, 1}, 16).has_value(), "P5 previewBBox refuses a result whose extent underflowed to zero");

    void p4() {

            const LogicalBox b = vertical ? LogicalBox{0,.5,1,0x1.0000000000002p-1} : LogicalBox{.5,0,0x1.0000000000002p-1,1};
            const auto both = damage(a, b);

            check(covers(outward(both, 1), b), "P4 fractional union preserves pixel coverage");
        }

        check(same(damage(prev, next), LogicalBox{-200,-90,370,230}), "P4 literal union of disjoint frames");
        check(same(damage(next, prev), damage(prev, next)), "P4 commutativity");
        check(same(damage(prev, prev), prev), "P4 idempotence");
        const LogicalBox inside{-190,-85,2,3};
        check(same(damage(prev, inside), prev), "P4 nested frame");
        for (double s : scales) {

            const auto both = damage(bbox(q, 96, 1, s), bbox(moved, 96, 1, s));
            auto a = q, b = moved; a.d.h += 96; b.d.h += 96;
            check(covers(outward(both, s), physical(a, s)) && covers(outward(both, s), physical(b, s)), "P4 damage covers old and new drawings");
        }

        for (double s : scales)

                check(same(PhysicalBox{b.x*s,b.y*s,b.w*s,b.h*s}, p), "P5 all components divide by scale exactly once");
                check(covers(outward(b, s), p), "P5 no one-pixel gap on round trip");
            }
        for (double s : {1., 1.1, 1.25, 1.3, 1.5, 2., 2.3, 3.})
            for (int i = -8; i <= 8; ++i)
                for (int width = 1; width <= 4; ++width)
                    for (double direction : {-INFINITY, INFINITY}) {
                        const PhysicalBox p{i / 3., -i / 3., std::nextafter(double(width), direction),
                                            std::nextafter(double(width), -direction)};
                        const auto b = preview(p, s);
                        check(same(PhysicalBox{b.x*s,b.y*s,b.w*s,b.h*s}, p), "P5 fractional components preserved within numeric error");
                        check(covers(outward(b, s), p), "P5 fractional edges preserve strict coverage");
                    }
        // Origins and edges one ulp either side of an integer: the division can round
        // across the integer, which is when the directed widening is needed.
        for (int k = 1; k <= 30; ++k) {
            const double s = 1 + k / 10.; // 1.1 .. 4.0, mostly inexact in binary
            for (int i = -4; i <= 4; ++i)
                for (double dx : {-INFINITY, INFINITY})
                    for (double dw : {-INFINITY, INFINITY}) {
                        const PhysicalBox p{std::nextafter(double(i), dx), std::nextafter(double(-i), -dx),
                                            std::nextafter(double(3), dw), std::nextafter(double(5), -dw)};
                        const auto b = preview(p, s);
                        check(covers(outward(b, s), p), "P5 ulp-adjacent origins and edges preserve strict coverage");
                    }
        }
        check(same(preview({251,-37,123,49}, 1.25), LogicalBox{200.8,-29.6,98.4,39.2}), "P5 literal fractional oracle");
    }
