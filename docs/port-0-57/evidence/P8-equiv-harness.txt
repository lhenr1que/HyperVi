    std::string s;
    for (auto& t : g_trace)

#!/usr/bin/env python3
"""P8 extraction.

extract.py calls SRC OUT MODE   MODE=old: plugin lines using m_dispatchers[...] / dispatchIfKnown(...) (K4-K8, K10-K12;
                                the K9 line `["mouse"]("0movewindow")` is out of scope and skipped);
                                MODE=new: plugin lines containing `KwAct::` (statements, not includes/comments).
extract.py old-fn SRC OUT       the plugin's own dispatchIfKnown definition, verbatim.
extract.py translator OSRC OUT  verbatim 0.56.2 DispatcherTranslator helpers + the 8 dispatchers the plugin used.
Each extracted plugin statement becomes `case K:` with a #line back to the real file.
"""
import re, sys

EXPECT = 10
ORDER = ["K4", "K5", "K6", "K7", "K8", "K10a", "K10b", "K11", "K12a", "K12b"]


def stmts(lines, pred):
    out, i, in_def = [], 0, False
    while i < len(lines):
        l = lines[i]
        if l.startswith("static void dispatchIfKnown("):
            in_def = True
        if in_def:
            if l.startswith("}"):
                in_def = False
            i += 1
            continue
        s = l.strip()
        if pred(l) and not s.startswith("//") and not s.startswith("#"):
            j = i
            while not lines[j].split("//")[0].rstrip().endswith(";"):
                j += 1
                assert j - i < 6, f"unterminated at {i+1}"
            out.append((i + 1, lines[i : j + 1]))
            i = j + 1
        else:
            i += 1
    return out


def main():
    mode = sys.argv[1]
    if mode == "calls":
        src, out, kind = sys.argv[2:5]
        lines = open(src, encoding="utf-8").read().split("\n")
        if kind == "old":
            pred = lambda l: ("m_dispatchers[" in l and '"mouse"' not in l) or "dispatchIfKnown(" in l
        else:
            pred = lambda l: "KwAct::" in l
        st = stmts(lines, pred)
        assert len(st) == EXPECT, f"{kind}: expected {EXPECT} statements, found {len(st)} at {[s[0] for s in st]}"
        def enclosing(ln):  # every enclosing control line, innermost first, up to the function header
            chain, ind = [], len(lines[ln - 1]) - len(lines[ln - 1].lstrip())
            for n in range(ln - 2, -1, -1):
                s = lines[n].strip()
                if s and not s.startswith("//") and len(lines[n]) - len(lines[n].lstrip()) < ind:
                    chain.append(s)
                    ind = len(lines[n]) - len(lines[n].lstrip())
                    if ind == 0:
                        break
            return " <- ".join(chain) or "?"
        with open(out + ".guards", "w") as g:
            for k, (ln, _) in enumerate(st):
                g.write(f"{ORDER[k]}: {enclosing(ln)}\n")
        with open(out, "w") as f:
            for k, (ln, body) in enumerate(st):
                f.write(f"    case {k}: {{ // {ORDER[k]}\n#line {ln} \"{src}\"\n" + "\n".join(body) + "\n    } break;\n")
        print(" ".join(f"{ORDER[k]}:{ln}" for k, (ln, _) in enumerate(st)))
    elif mode == "old-fn":
        src, out = sys.argv[2:4]
        lines = open(src, encoding="utf-8").read().split("\n")
        i = next(n for n, l in enumerate(lines) if l.startswith("static void dispatchIfKnown("))
        j = next(n for n in range(i, len(lines)) if lines[n].startswith("}"))
        open(out, "w").write(f"#line {i+1} \"{src}\"\n" + "\n".join(lines[i : j + 1]) + "\n")
        print(f"dispatchIfKnown {i+1}-{j+1}")
    elif mode == "translator":
        src, out = sys.argv[2:4]
        t = open(src, encoding="utf-8").read().split("\n")
        want = ["wrap", "windowFromArg", "resolveWorkspace", "exec", "setfloating", "settiled", "fullscreen",
                "movetoworkspacesilent", "movewindowpixel", "resizewindowpixel", "pin"]
        parts = []
        for name in want:
            rx = re.compile(r"^static [\w:<>]+ " + name + r"\(")
            i = next(n for n, l in enumerate(t) if rx.match(l))
            j = next(n for n in range(i, len(t)) if t[n].startswith("}"))
            parts.append(f"#line {i+1} \"{src}\"\n" + "\n".join(t[i : j + 1]))
        # CDispatcherTranslator::run, verbatim body (as a free function over a local map)
        i = next(n for n, l in enumerate(t) if l.startswith("SDispatchResult CDispatcherTranslator::run("))
        j = next(n for n in range(i, len(t)) if t[n].startswith("}"))
        run = "\n".join(t[i : j + 1]).replace("SDispatchResult CDispatcherTranslator::run(", "SDispatchResult translatorRun(")
        reg = []
        for n, l in enumerate(t):
            m = re.match(r'^\s*m_dispMap\["(\w+)"\]\s*=\s*::(\w+);', l)
            if m and m.group(1) in want:
                reg.append(f'#line {n+1} "{src}"\n' + l)
        assert len(reg) == 8, reg
        open(out, "w").write("\n\n".join(parts) + "\n\n"
                             "inline std::unordered_map<std::string, std::function<SDispatchResult(const std::string&)>> m_dispMap;\n"
                             + f"#line {i+1} \"{src}\"\n" + run + "\n"
                             "static void registerDispatchers() {\n" + "\n".join(reg) + "\n}\n")
        print("translator: " + " ".join(want) + f"; run {i+1}; {len(reg)} registrations")


main()

"""P8 log-policy oracle for the NEW side ('logs' mode: every scenario runs the statement 3 times in a row).
Expected, computed from the scenario alone (not from the implementation):
  K5/K7 with no focused window      -> 1 WARN  '<site> failed: <...>: no window (no_target)'
  K5 with a parse below 1x1          -> 1 ERR   'Invalid size (invalid_argument)'
  K12 with spawn pid 0 / no pid      -> 1 ERR   'Failed to start process ...' (execution_failed)
  otherwise a failed action result   -> info: 1 DEBUG, warn: 1 WARN, err: 1 ERR; silent/ok: none
Ordinary action failures must carry the host message ('fake <action> <info|warning|error>').
Three immediate repeats give exactly one line (the gate); after the window is rewound a 4th call gives a second line
ending '; 2 earlier failure(s) not logged'. No line when nothing failed."""

    else:
        print("unexpected line:", line); sys.exit(1)

bad = 0
for b in blocks:
    k, site = b["k"], SITES[b["k"]]
    exp = None  # (level, substring)
    if k in (8, 9):
        cmd = "kitty --title 'a b'" if k == 8 else "kwinish-ui menu <bar-window> --at 120,80"
        if b["spawn"] == "zero":
            exp = ("ERR", f"failed: Failed to start process {cmd} (execution_failed)")
        elif b["spawn"] == "none":
            exp = ("ERR", f"failed: Failed to start process. No closing bracket in exec rule. {cmd} (execution_failed)")
    elif k in (1, 3) and b["focus"] == "none":
        exp = ("WARN", ("resizeWindow" if k == 1 else "moveWindow") + ": no window (no_target)")
    elif k == 1 and b["parse"] == "tiny":
        exp = ("ERR", "Invalid size (invalid_argument)")
    elif b["res"] in LEVEL:
        fn = b["calls"].split(" ; ")[-1].split("(")[0]
        word = {"info": "info", "warn": "warning", "err": "error"}[b["res"]]
        exp = (LEVEL[b["res"]], f"fake {fn} {word} ({CODE[b['res']]})")
    logs = b["logs"]
    if exp is None:
        if logs:
            bad += 1; print("unexpected log:", b)
        continue
    if len(logs) != 2:
        bad += 1; print(f"expected exactly 2 log lines (first + summary), got {len(logs)}:", b); continue
    want = f"LOG[{exp[0]}] [kwinish-bars] {site} failed: "
    if not logs[0].startswith(want) or exp[1] not in logs[0] or "earlier failure" in logs[0]:
        bad += 1; print("wrong first log line:", logs[0], "| want prefix", want, "and", exp[1])
    if not logs[1].startswith(want) or exp[1] not in logs[1] or not logs[1].endswith("; 2 earlier failure(s) not logged"):
        bad += 1; print("wrong summary line:", logs[1])
    if k in (8, 9) and b["spawn"] in ("zero", "none") and "(execution_failed)" not in logs[0]:
        bad += 1; print("spawn failure without execution_failed code:", logs[0])
n_logged = sum(1 for b in blocks if b["logs"])
print(f"log oracle: {len(blocks)} scenarios, {n_logged} with first+summary lines, {bad} mismatches")
sys.exit(1 if bad or len(blocks) != 1350 else 0)

######## diff_audit.py ########
#!/usr/bin/env python3
"""P8 diff audit: every line changed in barDeco.cpp (BASE -> CAND) must be one of
  removed: an old extracted dispatcher statement, a line of the dispatchIfKnown definition (by position, + the blank
           line after it), a comment;
  added:   a new extracted KwAct statement, the hostActions.hpp include, a comment.
Within one hunk, an identical line removed and re-added (diff alignment, e.g. an unchanged `}`) cancels out.
Anything else (a new `return;`, a changed condition, a moved statement) fails, so the extracted statements stay
reachable exactly as before.  usage: diff_audit.py BASE CAND OLD_CALLS NEW_CALLS"""
import collections, difflib, sys

base, cand, oldc, newc = sys.argv[1:5]
b = open(base, encoding="utf-8").read().split("\n")
c = open(cand, encoding="utf-8").read().split("\n")


def stmt_lines(inc):
    return {l.rstrip("\n").strip() for l in open(inc, encoding="utf-8") if not l.startswith(("    case ", "#line", "    } break;"))}


old_ok, new_ok = stmt_lines(oldc), stmt_lines(newc)
i0 = next(n for n, l in enumerate(b) if l.startswith("static void dispatchIfKnown("))
j0 = next(n for n in range(i0, len(b)) if b[n].startswith("}"))
def_range = set(range(i0, j0 + 2 if b[j0 + 1].strip() == "" else j0 + 1))
import re


def single_statement(s):  # exactly one ';' outside string/char literals and comments, no jump keyword
    code = re.sub(r'"(?:[^"\\]|\\.)*"' + r"|'(?:[^'\\]|\\.)*'", '""', s).split("//")[0]
    return code.count(";") == 1 and code.rstrip().endswith(";") and not re.search(r"\b(return|goto|break|continue|throw)\b", code)


bad, counts = 0, collections.Counter()
for tag, i1, i2, j1, j2 in difflib.SequenceMatcher(None, b, c, autojunk=False).get_opcodes():
    if tag == "equal":
        continue
    rem = [(n, b[n].strip()) for n in range(i1, i2)]
    add = [c[n].strip() for n in range(j1, j2)]
    generic_rem, generic_add = collections.Counter(), collections.Counter()
    for n, s in rem:
        if s in old_ok: counts["-stmt"] += 1
        elif n in def_range: counts["-def"] += 1
        elif s.startswith("//"): counts["-comment"] += 1
        else: generic_rem[s] += 1
    for s in add:
        if s in new_ok and single_statement(s): counts["+stmt"] += 1
        elif s == '#include "hostActions.hpp"': counts["+include"] += 1
        elif s.startswith("//"): counts["+comment"] += 1
        else: generic_add[s] += 1
    moved = generic_rem & generic_add
    counts["realigned"] += sum(moved.values())
    for s, k in list((generic_rem - moved).items()) + list((generic_add - moved).items()):
        bad += k
        print(f"UNEXPLAINED change in hunk base {i1+1}-{i2} / cand {j1+1}-{j2}: '{s[:120]}'")
print("diff audit:", dict(counts), "unexplained:", bad)
sys.exit(1 if bad or counts["-stmt"] != 10 or counts["+stmt"] != 10 else 0)
